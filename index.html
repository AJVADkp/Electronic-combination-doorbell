02 FSM State Diagram & Transition Table 4-state Moore machine — 2-bit binary encoded (Q1, Q0)

State Transition Table

State Code [Q1 Q0] Input Condition Next State Bell Meaning
IDLE 00 B1 = 1 / B2 or B3 = 1 S1 / IDLE 0 Waiting for first button B1
S1 01 B2 = 1 / B1 or B3 = 1 S2 / IDLE 0 B1 valid; waiting for B2
S2 10 B3 = 1 / B1 or B2 = 1 UNLOCK / IDLE 0 B1+B2 valid; waiting for B3
UNLOCK 11 Next CLK (auto) IDLE 1 ★ RINGING Sequence accepted — bell rings 1 cycle!

Synthesized Boolean Equations (from K-Map)

State Decoder w_IDLE = ~Q1 · ~Q0 w_S1 = ~Q1 · Q0 w_S2 = Q1 · ~Q0 w_UNLOCK = Q1 · Q0
Next-State Excitation (D Flip-Flop Inputs) no_btn = ~(B1 + B2 + B3) D1 = ~RST · [(w_S1·B2) + (w_S2·B3) + (w_S2·no_btn)] D0 = ~RST · [(w_IDLE·B1) + (w_S1·no_btn) + (w_S2·B3)]
Moore Output BELL = Q1 · Q0

Bell is asserted HIGH exclusively in UNLOCK state (11). Rings for one clock cycle only.

03 Verilog HDL Implementation Synthesizable Verilog module — verilog/combination_doorbell.v 
// ================================================================
// Module: combination_doorbell
// Description: Electronic Combination Doorbell FSM (B1 → B2 → B3)
// Course: R25DER302 : Digital Electronics
// College: Christ College of Engineering (Autonomous), Irinjalakuda
// ================================================================
`timescale 1ns / 1ps

module combination_doorbell(
    input  wire clk,      // System Clock
    input  wire rst,      // Synchronous Reset (Active-High)
    input  wire b1, b2, b3, // Push-buttons
    output wire bell      // Bell Chime Output
);

    // ── 2-Bit State Encoding ──
    localparam [1:0] IDLE   = 2'b00,   // Waiting for B1
                       S1     = 2'b01,   // B1 valid, waiting for B2
                       S2     = 2'b10,   // B1+B2 valid, waiting for B3
                       UNLOCK = 2'b11;  // Accepted! Bell HIGH 1 cycle

    reg [1:0] state, next_state;

    // ── 01: Sequential State Register ──
    always @(posedge clk) begin
        if (rst)
            state <= IDLE;
        else
            state <= next_state;
    end

    // ── 02: Combinational Next-State Logic ──
    always @(*) begin
        next_state = state;  // default: hold state
        case (state)
            IDLE:   next_state = b1 ? S1 : IDLE;
            S1:     if (b2)        next_state = S2;
                    else if (b1 || b3) next_state = IDLE;
            S2:     if (b3)        next_state = UNLOCK;
                    else if (b1 || b2) next_state = IDLE;
            UNLOCK: next_state = IDLE;  // auto-return after 1 cycle
            default: next_state = IDLE;
        endcase
    end

    // ── 03: Moore Output — Bell rings only in UNLOCK ──
    assign bell = (state == UNLOCK);

endmodule

// Testbench: tb_combination_doorbell
// Tests: T1(B1→B2→B3 UNLOCK), T2(B1→B3 wrong), T3(B2 wrong start)
// ================================================================
`timescale 1ns / 1ps

module tb_combination_doorbell;
    reg  clk, rst, b1, b2, b3;
    wire bell;

    combination_doorbell uut (.clk(clk),.rst(rst),
                               .b1(b1),.b2(b2),.b3(b3),.bell(bell));

    always #5 clk = ~clk;

    initial begin
        $dumpfile("doorbell_simulation.vcd");
        $dumpvars(0, tb_combination_doorbell);

        clk=0; rst=1; b1=0; b2=0; b3=0;
        #20 rst=0;  // Release reset → IDLE

        // ── TEST 1: Correct sequence B1 → B2 → B3 ──
        @(posedge clk); b1=1; @(posedge clk); b1=0;  // → S1
        @(posedge clk); b2=1; @(posedge clk); b2=0;  // → S2
        @(posedge clk); b3=1; @(posedge clk); b3=0;  // → UNLOCK
        if (bell === 1'b1) $display("TEST 1 PASS: Bell=1 (UNLOCK)");
        @(posedge clk);  // → IDLE auto-reset
        #20;

        // ── TEST 2: Wrong sequence B1 → B3 ──
        @(posedge clk); b1=1; @(posedge clk); b1=0;  // → S1
        @(posedge clk); b3=1; @(posedge clk); b3=0;  // → IDLE (reject)
        if (bell === 1'b0) $display("TEST 2 PASS: Bell=0 (Rejected)");
        #20;

        // ── TEST 3: Wrong start B2 in IDLE ──
        @(posedge clk); b2=1; @(posedge clk); b2=0;  // stays IDLE
        if (bell === 1'b0) $display("TEST 3 PASS: Bell=0 (Ignored)");

        #50; $finish;
    end
endmodule
